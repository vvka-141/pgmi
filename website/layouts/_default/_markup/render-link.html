{{- /* Resolve GitHub-flavored relative .md links at build time so docs/*.md stay valid on GitHub while rendering correct site URLs here. - resolves to a published page -> internal site URL - escapes the published docs tree (e.g. repo-root README) -> GitHub source - unresolved within /docs -> warn (genuine broken link) */ -}} {{- $dest := .Destination -}} {{- $parts := split $dest "#" -}} {{- $pathPart := index $parts 0 -}} {{- $frag := "" -}} {{- if gt (len $parts) 1 }}{{ $frag = index $parts 1 }}{{ end -}} {{- if and (not (strings.Contains $pathPart "://")) (not (strings.HasPrefix $dest "/")) (not (strings.HasPrefix $dest "#")) (strings.HasSuffix (lower $pathPart) ".md") -}} {{- $rel := lower (strings.TrimSuffix ".md" $pathPart) -}} {{- /* A section page's Path IS its directory (/docs/advanced for docs/advanced/_index.md); only a leaf page needs path.Dir. Taking Dir unconditionally resolved every sibling link on an _index.md one level too high, so they silently became GitHub source links. */ -}} {{- $baseDir := cond (eq .Page.Kind "page") (path.Dir .Page.Path) .Page.Path -}} {{- $abs := path.Join $baseDir $rel -}} {{- with site.GetPage $abs -}} {{- $dest = .RelPermalink -}} {{- with $frag }}{{ $dest = printf "%s#%s" $dest . }}{{ end -}} {{- else -}} {{- /* .Page.Path is a leading-slash logical path (Hugo >= 0.123), so trim before matching -- without this every unresolved /docs link fell through to the GitHub-source branch and the warning never fired. */ -}} {{- $absNorm := strings.TrimPrefix "/" $abs -}} {{- $isDocsPath := or (eq $absNorm "docs") (strings.HasPrefix $absNorm "docs/") -}} {{- if not $isDocsPath -}} {{- $repoRel := path.Join (path.Dir (replace $.Page.File.Path "\\" "/")) $pathPart -}} {{- $dest = printf "%s/blob/main/%s" (site.Params.BookRepo | default "https://github.com/vvka-141/pgmi") $repoRel -}} {{- with $frag }}{{ $dest = printf "%s#%s" $dest . }}{{ end -}} {{- else -}} {{- warnf "render-link: broken internal .md link %q on page %s" $.Destination $.Page.Path -}} {{- end -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }}
